Créez un Google Apps Script gratuit qui sert de proxy — aucune clé API, aucun moteur CSE.

1. Ouvrez script.google.com → Nouveau projet
2. Effacez le code existant et collez le code ci-dessous
3. Cliquez Déployer → Nouveau déploiement
4. Type : Application Web — Accès : Tout le monde
5. Copiez l'URL du déploiement et collez-la ci-dessous 
function doGet(e) {
  var url = e.parameter.url;
  if (!url) return json({status:'error',msg:'no url'});
  try {
    var res = UrlFetchApp.fetch(
      'https://www.google.com/search?q=site:' + encodeURIComponent(url) + '&num=1&hl=fr',
      { headers: {'User-Agent':'Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/124'},
        followRedirects: true, muteHttpExceptions: true }
    );
    var html = res.getContentText();
    var noResult = /did not match any documents|n.a pas renvoy|aucun r.sultat|aucun document/i.test(html);
    var hasResult = /id="result-stats"|class="LC20lb|Environ.*r.sultat|About.*result/i.test(html);
    return json({status: noResult ? 'not-indexed' : hasResult ? 'indexed' : 'error'});
  } catch(err) {
    return json({status:'error', msg: err.message});
  }
}
function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
